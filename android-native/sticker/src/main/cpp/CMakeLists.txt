# :sticker 原生部分：ThorVG（只要 Lottie loader + CPU/SW 渲染器）+ 薄 JNI。
#
# ThorVG 官方用 meson；这里按 thorvg-sys 0.3.2+thorvg-1.1.0（桌面 lancat-slint 用的那份）的源码集合
# 和宏定义写成 CMake：renderer + cpu_engine + common + capi + raw loader + lottie loader，
# 不开 expressions / svg / png / fonts / threads / file-io / SIMD / partial render，渲染行为与桌面一致。
#
# 也用于 host（Windows/Linux）渲染测试：-DLC_HOST=ON 时不编 JNI，额外编 render_host 可执行文件。
cmake_minimum_required(VERSION 3.22.1)
project(lancatsticker CXX)

set(LC_THORVG_VERSION "1.1.0" CACHE STRING "ThorVG 版本（third_party/thorvg-<ver>）")
set(LC_THORVG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/thorvg-${LC_THORVG_VERSION}" CACHE PATH "ThorVG 源码目录")
option(LC_HOST "编 host 渲染测试，而不是 Android JNI 库" OFF)
option(LC_SIMD "开 ThorVG 的 NEON/AVX 光栅（默认关，与桌面一致）" OFF)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(TVG ${LC_THORVG_DIR}/src)

# config.h（与 thorvg-sys build.rs 的 write_config_h 对 features=["vendored","lottie"] 的输出一致）
set(LC_GEN ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(MAKE_DIRECTORY ${LC_GEN})
file(WRITE ${LC_GEN}/config.h.tmp
"/* Generated by :sticker CMakeLists.txt */\n#pragma once\n\n#define THORVG_CAPI_BINDING_SUPPORT 1\n#define THORVG_CPU_ENGINE_SUPPORT 1\n#define THORVG_VERSION_STRING \"${LC_THORVG_VERSION}\"\n#define THORVG_LOTTIE_LOADER_SUPPORT 1\n")
configure_file(${LC_GEN}/config.h.tmp ${LC_GEN}/config.h COPYONLY)

file(GLOB TVG_SRC CONFIGURE_DEPENDS
  ${TVG}/renderer/*.cpp
  ${TVG}/renderer/cpu_engine/*.cpp
  ${TVG}/common/*.cpp
  ${TVG}/bindings/capi/*.cpp
  ${TVG}/loaders/raw/*.cpp
  ${TVG}/loaders/lottie/*.cpp)
list(FILTER TVG_SRC EXCLUDE REGEX "tvgLottieExpressions\\.cpp$")

add_library(thorvg STATIC ${TVG_SRC})
target_include_directories(thorvg
  PUBLIC ${LC_GEN} ${LC_THORVG_DIR}/inc ${TVG}/bindings/capi
  PRIVATE ${TVG}/renderer ${TVG}/renderer/cpu_engine ${TVG}/common
          ${TVG}/loaders/raw ${TVG}/loaders/lottie ${TVG}/loaders/lottie/rapidjson)
target_compile_definitions(thorvg PUBLIC TVG_STATIC PRIVATE _DEFAULT_SOURCE)
if(MSVC)
  target_compile_definitions(thorvg PRIVATE NOMINMAX)
  target_compile_options(thorvg PRIVATE /EHsc /W0)
else()
  # 与 meson / thorvg-sys 一致
  target_compile_options(thorvg PRIVATE -fno-exceptions -fno-rtti -fno-stack-protector -fno-math-errno
    -fno-unwind-tables -fno-asynchronous-unwind-tables -w)
  if(WIN32)
    target_compile_definitions(thorvg PRIVATE NOMINMAX)
  endif()
endif()
# Debug 包里 ThorVG 也用 -O2：-O0 下单帧要慢好几倍，聊天里播放会卡
if(NOT MSVC AND CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_options(thorvg PRIVATE -O2)
endif()
set_target_properties(thorvg PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(LC_SIMD)
  # 1.1.0 用 *_VECTOR_SUPPORT，1.1.3 起改名为 THORVG_NEON_SUPPORT / THORVG_AVX_SUPPORT
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
    target_compile_definitions(thorvg PRIVATE THORVG_NEON_VECTOR_SUPPORT THORVG_NEON_SUPPORT)
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    target_compile_definitions(thorvg PRIVATE THORVG_AVX_VECTOR_SUPPORT)
    if(NOT MSVC)
      target_compile_options(thorvg PRIVATE -mavx)
    endif()
  endif()
endif()

add_library(lc_lottie STATIC lc_lottie.cpp)
target_link_libraries(lc_lottie PUBLIC thorvg)
target_compile_definitions(lc_lottie PRIVATE LC_THORVG_VERSION="${LC_THORVG_VERSION}")
set_target_properties(lc_lottie PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_compile_options(lc_lottie PRIVATE -fno-exceptions -fno-rtti)
endif()

if(LC_HOST)
  add_executable(render_host ${CMAKE_CURRENT_SOURCE_DIR}/../../../host/render_host.cpp)
  target_include_directories(render_host PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(render_host PRIVATE lc_lottie)
else()
  add_library(lancatsticker SHARED lc_lottie_jni.cpp)
  target_link_libraries(lancatsticker PRIVATE lc_lottie jnigraphics log)
  target_compile_options(lancatsticker PRIVATE -fno-exceptions -fno-rtti)
  # 只导出 JNI 符号；16 KB 页对齐（Android 15+ 设备要求，NDK r27 还不是默认值，r28 起默认）
  target_link_options(lancatsticker PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
endif()
